CREATE TABLE FOOTER_SETTINGS (
  ID NUMBER(10) PRIMARY KEY,
  NAME VARCHAR2(100) NOT NULL,
  ADDRESS VARCHAR2(255) NOT NULL,
  CLINIC_NAME VARCHAR2(100) NOT NULL,
  PHONE VARCHAR2(50) NOT NULL,
  EMAIL VARCHAR2(100) NOT NULL,
  LOCATION_URL VARCHAR2(255) NOT NULL,
  SCHEDULES CLOB NOT NULL,
  FAMILY_SITES CLOB NOT NULL
);

COMMENT ON TABLE FOOTER_SETTINGS IS '푸터(하단) 영역 설정 데이터';
COMMENT ON COLUMN FOOTER_SETTINGS.ID IS '설정 고유번호 (1 고정)';
COMMENT ON COLUMN FOOTER_SETTINGS.SCHEDULES IS '진료시간 JSON 배열';
COMMENT ON COLUMN FOOTER_SETTINGS.FAMILY_SITES IS '패밀리사이트 JSON 배열';

ALTER TABLE FOOTER_SETTINGS ADD CONSTRAINT CHK_SCHEDULES_JSON CHECK (SCHEDULES IS JSON);
ALTER TABLE FOOTER_SETTINGS ADD CONSTRAINT CHK_FAMILY_SITES_JSON CHECK (FAMILY_SITES IS JSON);

CREATE TABLE CONSULTS (
  ID NUMBER(10) GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  NAME VARCHAR2(50) NOT NULL,
  PHONE VARCHAR2(20) NOT NULL,
  PASSWORD VARCHAR2(255) NOT NULL,
  USER_ID VARCHAR2(50) NOT NULL,
  DEPARTMENT VARCHAR2(50) NOT NULL,
  TITLE VARCHAR2(200) NOT NULL,
  CONTENT CLOB NOT NULL,
  STATUS VARCHAR2(20) DEFAULT '대기중',
  CREATED_AT TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

COMMENT ON TABLE CONSULT IS '온라인 상담(게시판) 테이블';
COMMENT ON COLUMN CONSULT.ID IS '상담글 고유번호';
COMMENT ON COLUMN CONSULT.NAME IS '작성자 이름';
COMMENT ON COLUMN CONSULT.PHONE IS '작성자 연락처';
COMMENT ON COLUMN CONSULT.PASSWORD IS '비회원 글 확인용 비밀번호';
COMMENT ON COLUMN CONSULT.USER_ID IS '회원 아이디 (비회원시 NULL)';
COMMENT ON COLUMN CONSULT.DEPARTMENT IS '진료 과목';
COMMENT ON COLUMN CONSULT.TITLE IS '상담 제목';
COMMENT ON COLUMN CONSULT.CONTENT IS '상담 내용 본문';
COMMENT ON COLUMN CONSULT.STATUS IS '상담 처리 상태 (대기중/답변완료)';
COMMENT ON COLUMN CONSULT.CREATED_AT IS '글 작성 시간';
